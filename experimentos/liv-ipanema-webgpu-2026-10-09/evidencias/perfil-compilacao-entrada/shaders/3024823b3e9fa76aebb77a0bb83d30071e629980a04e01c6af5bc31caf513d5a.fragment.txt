// structs

layout( location = 0 ) out vec4 fragColor;



// uniforms

layout( std140 ) uniform object {
	float nodeUniform0;
	mat4 nodeUniform3;
};


// varyings


// vars
vec4 DiffuseColor;
vec4 Output;
vec4 nodeVar0;

// codes


void main() {

	// flow
	// code

	DiffuseColor = vec4( 0.0, 0.0, 0.0, 1.0 );
	DiffuseColor.w = ( DiffuseColor.w * nodeUniform0 );
	nodeVar0 = max( vec4( DiffuseColor.xyz, DiffuseColor.w ), 0.0 );
	Output = nodeVar0;

	// result
	fragColor = nodeVar0;

}
