vec3 positionLocal;

// attributes
layout( location = 0 ) in vec3 position;

// vars

// codes


void main() {

	// transforms
	

	// flow
	// code

	positionLocal = position;
	nodeVar0 = mat4( nodeAttribute0, nodeAttribute1, nodeAttribute2, nodeAttribute3 );
	positionLocal = ( nodeVar0 * vec4( positionLocal, 1.0 ) ).xyz;
	normalLocal = normal;
	normalLocal = normalize( ( transpose( inverse( mat3( nodeVar0[ 0 ].xyz, nodeVar0[ 1 ].xyz, nodeVar0[ 2 ].xyz ) ) ) * normalLocal ) );
	v_normalViewGeometry = normalize( ( cameraViewMatrix * vec4( ( nodeUniform9 * normalLocal ), 0.0 ) ).xyz );
	modelViewMatrix = ( cameraViewMatrix * nodeUniform13 );
	v_positionView = ( modelViewMatrix * vec4( positionLocal, 1.0 ) ).xyz;
	v_positionViewDirection = ( - v_positionView );
	v_positionWorld = ( nodeUniform13 * vec4( positionLocal, 1.0 ) ).xyz;
	VERTEX_nodeVar262 = ( cameraProjectionMatrix * vec4( v_positionView, 1.0 ) );
	VERTEX_v_modelViewProjection = VERTEX_nodeVar262;

	// result
	gl_Position = VERTEX_v_modelViewProjection;

	gl_PointSize = 1.0;

}
